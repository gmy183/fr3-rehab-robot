----- /start_xhand.sh -----
#!/usr/bin/bash

# 定义锁文件, 防止重复多次执行
LOCK_FILE="/tmp/xhand_controller.lock"

# 检查是否已有锁文件，表示程序已在运行
if [ -f "$LOCK_FILE" ]; then
    # 使用 notify-send 弹出提示框告知用户程序正在运行
    notify-send "提示" "XHAND 已经在运行，请勿重复启动。"
    exit 1  # 退出脚本
else
    # 创建锁文件
    echo $$ > "$LOCK_FILE"
    trap 'rm -f "$LOCK_FILE"' EXIT
fi

notify-send "提示" "XHAND 正在启动, 大约需要 1 分钟！"

# 定义日志文件路径
LOG_FILE_1="/tmp/xhand_setup_1.log"

# 检查日志文件是否存在，如果不存在则创建它
if [ ! -f "$LOG_FILE_1" ]; then
    touch "$LOG_FILE_1"

    # 检查创建日志文件是否成功
    if [ $? -eq 0 ]; then
        chmod 666 "$LOG_FILE_1"  # 赋予读写权限，确保日志文件可写
        echo "$(date): 1日志文件已创建" | tee -a "$LOG_FILE_1"
    else
        LOG_FILE_1=""  # 创建失败，将LOG_FILE_1指向空
        echo "$(date): 日志文件创建失败，LOG_FILE_1已指向空" | tee -a /dev/stderr
    fi
fi

# 检查Redis是否正在运行（检查端口 6379）
if ! nc -z localhost 6379; then
    echo "$(date): Redis 未运行，正在启动 Redis..." | tee -a "$LOG_FILE_1"
    nohup /usr/local/redis7.4/bin/redis-server >> "$LOG_FILE_1" 2>&1 &
    echo "$(date): Redis 已启动" | tee -a "$LOG_FILE_1"
else
    echo "$(date): Redis 已经在运行" | tee -a "$LOG_FILE_1"
fi

## 安装 xhand_controller
#echo "$(date): 正在安装 xhand_controller..." | tee -a "$LOG_FILE_1"
#/usr/bin/pip3 install /usr/local/xhand_controller/xhand_controller/. >> "$LOG_FILE_1" 2>&1
#if [ $? -eq 0 ]; then
#    echo "$(date): xhand_controller 安装完成" | tee -a "$LOG_FILE_1"
#else
#    echo "$(date): xhand_controller 安装失败" | tee -a "$LOG_FILE_1"
#    exit 1
#fi

# sudo 启动Python应用, pkexec 执行获取密码输入框
echo "$(date): 启动pkexec脚本..." | tee -a "$LOG_FILE_1"
pkexec bash start_pkexec.sh >> "$LOG_FILE_1" 2>&1
if [ $? -eq 0 ]; then
    echo "$(date): pkexec 启动完成" | tee -a "$LOG_FILE_1"
    notify-send "提示" "XHAND 启动完成！"
else
    echo "$(date): pkexec 启动失败" | tee -a "$LOG_FILE_1"
    notify-send "错误" "XHAND 启动失败，尝试重新启动！"
    exit 1
fi

sleep 3
# 使用系统默认浏览器打开网址 http://127.0.0.1:1888
xdg-open http://127.0.0.1:1888
echo "$(date): 打开网址 http://127.0.0.1:1888" | tee -a "$LOG_FILE_1"


## sudo 启动监视浏览器脚本
#echo "$(date): 启动监视浏览器脚本..." | tee -a "$LOG_FILE_1"
#pkexec bash monitor_browser_pkexec.sh >> "$LOG_FILE_1" 2>&1
